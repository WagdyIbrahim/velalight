const $  = s => document.querySelector(s);
const $$ = s => [...document.querySelectorAll(s)];
let LANG = localStorage.getItem("vl-lang") || "ar";
const t  = o => o ? (o[LANG] ?? o.ar) : "";

const IS_PRODUCT = location.pathname.includes("product");

function starsHTML(r){ const f=Math.round(r); return "★".repeat(f)+"☆".repeat(5-f); }
function toast(msg){
  const el=$("#toast"); if(!el) return;
  el.textContent=msg; el.classList.add("show");
  clearTimeout(el._t); el._t=setTimeout(()=>el.classList.remove("show"),2600);
}

/* ─────────────── اللغة ─────────────── */
function applyLang(){
  document.documentElement.lang = LANG==="ar"?"ar":"en";
  document.documentElement.dir  = LANG==="ar"?"rtl":"ltr";
  $$("[data-ar]").forEach(el=>{ const v=el.getAttribute(LANG==="ar"?"data-ar":"data-en"); if(v!==null) el.textContent=v; });
  $$("[data-ar-ph]").forEach(el=>{ el.placeholder = el.getAttribute(LANG==="ar"?"data-ar-ph":"data-en-ph")||""; });
  const lt=$("#langToggle"); if(lt) lt.textContent = LANG==="ar"?"EN":"عربي";
  renderAll();
}
function toggleLang(){ LANG = LANG==="ar"?"en":"ar"; localStorage.setItem("vl-lang",LANG); applyLang(); }

function cardHTML(p){
  const off = p.old ? Math.round((1-p.price/p.old)*100) : 0;
  return `
  <article class="p-card">
    <a class="p-img" href="product.html?id=${p.id}">
      <img src="${p.img[0]}" alt="${t(p.name)}" loading="lazy" onerror="this.src='${FALLBACK_IMG}'">
      ${off?`<span class="off">-${off}%</span>`:""}

    <div class="p-body">
      <span class="p-cat">${t(CAT_NAMES[p.cat])}</span>
      <h3 class="p-name"><a href="product.html?id=${p.id}">${t(p.name)}</a></h3>
      <div class="p-rate">${starsHTML(p.rating)} <b>${p.rating}/5</b></div>

}

let cart = JSON.parse(localStorage.getItem("vl-cart")||"[]");
const saveCart = () => localStorage.setItem("vl-cart", JSON.stringify(cart));
function addToCart(id,scent,qty=1){
  const p=PRODUCTS.find(x=>x.id===id); if(!p) return;
  scent = scent || t(SCENTS[0]);
  const f=cart.find(i=>i.id===id&&i.scent===scent);
  f ? f.qty+=qty : cart.push({id,scent,qty});
  saveCart(); renderCart();
  toast(LANG==="ar"?"تمت الإضافة للسلة ✅":"Added to cart ✅");
}
function quickAdd(id){ addToCart(id); }
function renderCart(){
  const box=$("#cartItems"); if(!box) return;
  $("#cartCount").textContent = cart.reduce((s,i)=>s+i.qty,0);
  if(!cart.length){
    box.innerHTML = `<p class="cart-empty">${LANG==="ar"?"السلة فاضية... ابدأ التسوق 🤎":"Your cart is empty... start shopping 🤎"}</p>`;
  } else {
    box.innerHTML = cart.map((i,idx)=>{
      const p=PRODUCTS.find(x=>x.id===i.id);
      return `<div class="ci">
        <img src="${p.img[0]}" alt="${t(p.name)}" onerror="this.src='${FALLBACK_IMG}'">
        <div class="ci-info"><strong>${t(p.name)}</strong><span class="ci-scent">🌸 ${i.scent}</span>
          <div class="ci-qty"><button onclick="ciQty(${idx},-1)">−</button><b>${i.qty}</b><button onclick="ciQty(${idx},1)">+</button><span class="muted">${fmt(p.price*i.qty)}</span></div>
        </div>
        <button class="ci-rm" onclick="ciRemove(${idx})" aria-label="Remove">✖</button></div>`;
    }).join("");
  }
  $("#cartTotal").textContent = fmt(cart.reduce((s,i)=>s+PRODUCTS.find(p=>p.id===i.id).price*i.qty,0));
}
function ciQty(i,d){ cart[i].qty+=d; if(cart[i].qty<1) cart.splice(i,1); saveCart(); renderCart(); }
function ciRemove(i){ cart.splice(i,1); saveCart(); renderCart(); }

function closeCart(){ $("#cartDrawer").classList.remove("open"); $("#overlay").classList.remove("show"); document.body.style.overflow=""; }

function checkout(){
  if(!cart.length) return toast(LANG==="ar"?"السلة فاضية 🤎":"Your cart is empty 🤎");
  const name=$("#dl-name").value.trim(), phone=$("#dl-phone").value.trim(), addr=$("#dl-address").value.trim();
  if(!name||!phone) return toast(LANG==="ar"?"من فضلك اكتبي الاسم ورقم الهاتف 🙏":"Please enter your name and phone 🙏");
  const total = cart.reduce((s,i)=>s+PRODUCTS.find(p=>p.id===i.id).price*i.qty,0);
  const L = LANG==="ar";
  let msg = L?"🕯️ *طلب جديد من موقع VelaLight*\n":"🕯️ *New order from VelaLight website*\n";
  msg += (L?"👤 الاسم: ":"👤 Name: ")+name+"\n"+(L?"📞 الهاتف: ":"📞 Phone: ")+phone+"\n";
  if(addr) msg += (L?"📍 العنوان: ":"📍 Address: ")+addr+"\n";
  msg += L?"\n🛍️ *المنتجات:*\n":"\n🛍️ *Items:*\n";
  cart.forEach(i=>{ const p=PRODUCTS.find(x=>x.id===i.id); msg+=`• ${t(p.name)} (${i.scent}) ×${i.qty} — ${fmt(p.price*i.qty)}\n`; });
  msg += (L?"\n💰 الإجمالي: ":"\n💰 Total: ")+fmt(total)+(L?" (بخلاف الشحن)":" (excl. shipping)");
  msg += L?"\n💳 الدفع: InstaPay مقدمًا + الشحن كاش عند الاستلام":"\n💳 Payment: InstaPay in advance + shipping cash on delivery";
  const orders=JSON.parse(localStorage.getItem("vl-orders")||"[]");
  orders.unshift({date:new Date().toISOString().slice(0,10), total, items:cart.map(i=>t(PRODUCTS.find(p=>p.id===i.id).name))});
  localStorage.setItem("vl-orders",JSON.stringify(orders));
  window.open(`https://wa.me/${WHATSAPP}?text=${encodeURIComponent(msg)}`,"_blank");
}

/* المعرض: كل الصور تظهر + أسهم + سحب */
function renderGallery(){
  const p=curProduct; if(!p) return;
  curImg=0;
  $("#pdThumbs").innerHTML=p.img.map((src,i)=>
    `<img src="${src}" alt="${t(p.name)} ${i+1}" class="${i===0?'on':''}" loading="lazy" onerror="this.src='${FALLBACK_IMG}'" data-i="${i}">`).join("");
  $$("#pdThumbs img").forEach(th=>th.onclick=()=>showImg(+th.dataset.i));
  showImg(0);
}
function showImg(i){
  const p=curProduct; if(!p) return;
  curImg=(i+p.img.length)%p.img.length;
  const main=$("#pdMainImg");
  main.src=p.img[curImg]; main.onerror=()=>{main.src=FALLBACK_IMG;};
  main.alt=t(p.name);
  $$("#pdThumbs img").forEach((th,idx)=>th.classList.toggle("on",idx===curImg));
}
function bindGallery(){
  $("#galPrev")?.addEventListener("click",()=>showImg(curImg-1));
  $("#galNext")?.addEventListener("click",()=>showImg(curImg+1));
  const main=$(".pd-main");
  if(main){ let x0=null;
    main.addEventListener("touchstart",e=>x0=e.touches[0].clientX,{passive:true});
    main.addEventListener("touchend",e=>{ if(x0===null)return;
      const dx=e.changedTouches[0].clientX-x0;
      if(Math.abs(dx)>40) dx>0?showImg(curImg-1):showImg(curImg+1);
      x0=null; },{passive:true});
  }
}

/* ─────────────── التقييمات ─────────────── */
function getAllReviews(){
  const saved = curProduct ? JSON.parse(localStorage.getItem("vl-reviews-"+curProduct.id)||"[]") : [];
  return [...saved, ...DEFAULT_REVIEWS];
}
function renderReviews(){
  const list=$("#reviewList"); if(!list||!curProduct) return;
  const all=getAllReviews();
  const avg=(all.reduce((s,r)=>s+r.stars,0)/all.length).toFixed(1);
  $("#revAvg").textContent=`${avg}/5`;
  $("#revAvgStars").textContent=starsHTML(+avg);
  list.innerHTML=all.map(r=>{
    const nm=typeof r.name==="object"?t(r.name):r.name;
    const tx=typeof r.text==="object"?t(r.text):r.text;
    return `<div class="review-card"><div class="rc-head">
      <span class="rc-avatar">${nm.charAt(0)}</span>
      <div><div class="rc-name">${nm} ${r.saved?'✨':''}</div><div class="rc-stars">${starsHTML(r.stars)} ${r.stars}/5</div></div>
    </div><p>${tx}</p></div>`;
  }).join("");
}
function bindReviewForm(){
  const pick=$("#starPick"); if(!pick) return;
  const paint=()=>pick.querySelectorAll("button").forEach(x=>x.classList.toggle("on",+x.dataset.v<=pickedStars));
  pick.querySelectorAll("button").forEach(b=>b.onclick=()=>{ pickedStars=+b.dataset.v; paint(); });
  paint();
  $("#reviewForm")?.addEventListener("submit",e=>{
    e.preventDefault();
    const name=$("#rvName").value.trim(), text=$("#rvText").value.trim();
    if(!name||!text) return;
    const saved=JSON.parse(localStorage.getItem("vl-reviews-"+curProduct.id)||"[]");
    saved.unshift({name,text,stars:pickedStars,saved:true});
    localStorage.setItem("vl-reviews-"+curProduct.id,JSON.stringify(saved));
    e.target.reset(); pickedStars=5; paint(); renderReviews();
    toast(LANG==="ar"?"شكرًا لتقييمك 🤎":"Thank you for your review 🤎");
  });
}

    const sel=$("#pdScent");
    if(sel){ const c=sel.selectedIndex; sel.innerHTML=SCENTS.map(sc=>`<option>${t(sc)}</option>`).join(""); sel.selectedIndex=Math.max(0,c); }
  }
}
